סוגי משתמשים


המערכת תומכת בשני סוגי משתמשים עם ממשקים שונים:

async Task Login()
{
    TunersDB T = new TunersDB();
    Tuners tuners = await T.Login(email, password);
    
    if (tuners != null)
    {
        if (DateTime.Now < tuners.bantime)
        {
            TimeSpan remaining = tuners.bantime - DateTime.Now;
            if (remaining.TotalDays > 3650)
            {
                message = "User is permanently banned";
            }
            else if (remaining.TotalDays >= 1)
            {
                message = $"User is banned for {Math.Ceiling(remaining.TotalDays)} more days";
            }
            else if (remaining.TotalHours >= 1)
            {
                message = $"User is banned for {Math.Ceiling(remaining.TotalHours)} more hours";
            }
            else
            {
                message = $"User is banned for {Math.Ceiling(remaining.TotalMinutes)} more minutes";
            }
            email = "";
            password = "";
        }
        else
        {
            tuners.picture = null;
            await MySession.SetAsync("Tuner", tuners);
            navigationManager.NavigateTo("/Chatsss");
        }
    }
    else
    {
        message = "Invalid email or password";
        email = "";
        password = "";
    }
}

public async Task SendEmail()
{
    if (string.IsNullOrWhiteSpace(email))
    {
        errorMsg = "Please enter your email.";
        return;
    }

    List<string> existingEmails = await TunersDB.GetAllEmailsAsync();
    if (existingEmails.Contains(email))
    {
        errorMsg = "This email is already in use.";
        return;
    }

    try
    {
        Random rnd = new Random();
        confnum = rnd.Next(100000, 999999);
        await EmailService.SendMessage(email, confnum);
        step = 2;
        errorMsg = "";
    }
    catch (Exception ex)
    {
        errorMsg = "Failed to send email: " + ex.Message;
    }
}

public async Task CreateAccount()
{
    if (string.IsNullOrWhiteSpace(username))
    {
        errorMsg = "Username is required.";
        return;
    }

    if (password != passwordconf)
    {
        passwordconf = "";
        errorMsg = "Confirm Password must match Password.";
        return;
    }

    if (!ValidatePassword(password))
    {
        password = "";
        passwordconf = "";
        return;
    }

    Tuners tuner = await TunersDB.Reg(username, email, password);
    if (tuner == null)
    {
        errorMsg = "Email already in use — please go back and try another.";
        return;
    }

    if (!string.IsNullOrEmpty(pictureBase64))
    {
        await TunersDB.UpdatePictureAsync(tuner.Id, pictureBase64);
        tuner.picture = null;
    }

    await MySession.SetAsync("Tuner", tuner);
    navigationManager.NavigateTo("/Chatsss");
}

private async Task BanUser(int reportId, int reportedUserId, TimeSpan duration)
{
    bool success = await Ban(reportId, reportedUserId, duration);

    if (success)
    {
        Reports = await ReportDB.GetUsersReportsAsync(reportedUserId);
        await ReportDB.DeleteReportsListAsync(Reports);
        Report.RemoveAll(r => r.reported == reportedUserId);
        showBanDurations = false;
        selectedReportId = -1;
        StateHasChanged();
    }
}

private async Task<bool> Ban(int reportId, int reportedUserId, TimeSpan duration)
{
    DateTime banUntil = DateTime.Now.Add(duration);
    int t = await Tune.SetBanTime(reportedUserId, banUntil);
    int r = await ReportDB.DeleteReport(reportId);
    return r != 0 && t != 0;
}

private bool ValidatePassword(string pwd)
{
    if (pwd.Length < 6)
    {
        errorMsg = "Password must be at least 6 characters long.";
        return false;
    }
    if (!pwd.Any(char.IsDigit))
    {
        errorMsg = "Password must contain at least 1 number.";
        return false;
    }
    if (!pwd.Any(ch => !char.IsLetterOrDigit(ch)))
    {
        errorMsg = "Password must contain at least 1 special character.";
        return false;
    }
    return true;
}

בדיקות:
- אורך מינימלי: 6 תווים
- חייבת להכיל לפחות ספרה אחת (0-9)
- חייבת להכיל לפחות תו מיוחד אחד (!@#$%^&* וכו')

סיכום Validation


המערכת מבצעת בדיקות תקינות ב-4 רמות:

1. Client-Side Validation:
   - בדיקות מיידיות בממשק המשתמש
   - מונעות שליחת נתונים לא תקינים לשרת
   - משפרות את חווית המשתמש (תגובה מהירה)

2. Server-Side Validation:
   - בדיקות נוספות בשרת
   - מונעות עקיפה של בדיקות הלקוח
   - מבטיחות תקינות הנתונים במסד הנתונים

3. Business Logic Validation:
   - בדיקות לוגיות (Cooldown, קטגוריות, וכו')
   - מבטיחות שהמערכת פועלת לפי הכללים העסקיים

4. Security Validation:
   - הגנה מפני SQL Injection
   - הגנה מפני גישה לא מורשית
   - בדיקת הרשאות (role)
