2. מימוש ChatHub

namespace Tune.site.Hubs
{
    public class ChatHub : Hub
    {
        private readonly MessagesDB _messagesDB;

        public ChatHub(MessagesDB messagesDB)
        {
            _messagesDB = messagesDB;
        }

        public async Task JoinChatGroup(int chatId)
        {
            await Groups.AddToGroupAsync(Context.ConnectionId, $"Chat_{chatId}");
        }

        public async Task SendMessage(int chatId, int userId, string text, DateTime time)
        {
            await Clients.Group($"Chat_{chatId}")
                .SendAsync("ReceiveMessage", chatId, userId, text, time);
        }

        public async Task DeleteMessage(int messageId, int chatId)
        {
            await _messagesDB.Delete(messageId);

            var remainingMessages = await _messagesDB.SelectMessagesByChatId(chatId);
            var newLastMessage = remainingMessages.OrderByDescending(m => m.Time).FirstOrDefault();

            await Clients.Group($"Chat_{chatId}")
                .SendAsync("MessageDeleted", messageId,
                    newLastMessage?.text ?? "",
                    newLastMessage?.Time ?? DateTime.MinValue);
        }

        public async Task UpdateMessage(int messageId, int chatId, string newText)
        {
            await _messagesDB.UpdateMessageText(messageId, newText);

            var allMessages = await _messagesDB.SelectMessagesByChatId(chatId);
            var lastMessage = allMessages.OrderByDescending(m => m.Time).FirstOrDefault();

            await Clients.Group($"Chat_{chatId}")
                .SendAsync("MessageUpdated",
                    messageId,
                    newText,
                    lastMessage?.id == messageId ? newText : lastMessage?.text ?? "");
        }
    }
}

הסבר על הפעולות:

JoinChatGroup:
- מוסיפה את החיבור הנוכחי לקבוצה של צ'אט מסוים
- כל צ'אט הוא קבוצה נפרדת (Chat_1, Chat_2, וכו')
- רק משתמשים בקבוצה יקבלו הודעות של הצ'אט הזה

SendMessage:
- שולחת הודעה לכל המשתמשים בקבוצת הצ'אט
- משתמשת ב-Clients.Group כדי לשלוח רק למשתמשים הרלוונטיים
- מעבירה את פרטי ההודעה (מזהה צ'אט, מזהה משתמש, טקסט, זמן)

DeleteMessage:
- מוחקת הודעה ממסד הנתונים
- מוצאת את ההודעה האחרונה החדשה
- מודיעה לכל המשתמשים בצ'אט על המחיקה

UpdateMessage:
- מעדכנת הודעה במסד הנתונים
- מוצאת את ההודעה האחרונה (למקרה שההודעה המעודכנת היא האחרונה)
- מודיעה לכל המשתמשים בצ'אט על העדכון

public async Task<TimeSpan?> InsertGetObjAsync(Reports R)
{
    if (!IsValidCategory(R.report))
        return null;

    TunersDB s = new TunersDB();
    Tuners t = await s.SelectUserID(R.reported);

    if (t == null || t.bantime > DateTime.UtcNow)
        return null;

    Reports existing = await GetRecentReport(R.reporter, R.reported);
    if (existing != null)
    {
        TimeSpan elapsed = DateTime.UtcNow - existing.created_at;
        TimeSpan cooldown = TimeSpan.FromHours(24);
        if (elapsed < cooldown)
            return cooldown - elapsed;
    }

    Dictionary<string, object> fillValues = new Dictionary<string, object>()
    {
        { "reporter", R.reporter },
        { "reported", R.reported },
        { "report", R.report },
        { "created_at", DateTime.UtcNow }
    };

    await base.InsertGetObjAsync(fillValues);
    return null;
}

הסבר:

2. Custom Music API - שירות נתוני מוזיקה

המערכת משתמשת ב-API מותאם אישית (project12api) לקבלת נתוני האזנות מוזיקליות.

נקודות קצה (Endpoints):

GET /api/UserGenres/{email}
- מחזיר את וקטור הז'אנרים של משתמש אחד
- משמש לקבלת העדפות המוזיקה של משתמש

POST /api/UserGenres/batch
- מקבל רשימת אימיילים
- מחזיר וקטורי ז'אנרים של כל המשתמשים
- משמש לאלגוריתם KNN למציאת משתמשים דומים


מימוש ה-API Controller:

[ApiController]
[Route("api/[controller]")]
public class UserGenresController : ControllerBase
{
    [HttpGet("{email}")]
    public async Task<IActionResult> GetGenres(string email)
    {
        ReplayInfoDB RI = new ReplayInfoDB();
        List<ReplayInfo> R = await RI.GetUserGenres(email);
        
        if (R == null) 
            return NotFound("User not found in API");
            
        var result = R;
        return Ok(result);
    }

    [HttpPost("batch")]
    public async Task<IActionResult> GetGenresBatch([FromBody] List<string> emails)
    {
        UsersDB U = new UsersDB();
        List<object> result = new List<object>();
        
        foreach (string email in emails)
        {
            ReplayInfoDB RI = new ReplayInfoDB();
            List<ReplayInfo> R = await RI.GetUserGenres(email);
            
            if (R.Count == 0) continue;
            
            var genreVector = R;
            result.Add(new
            {
                Email = email,
                Genres = genreVector
            });
        }
        
        return Ok(result);
    }
}

הסבר:

GetGenres (GET):
- מקבל אימייל של משתמש
- שולף את נתוני ההאזנות שלו (ז'אנרים וכמויות)
- מחזיר רשימה של ReplayInfo (ז'אנר + כמות)
- אם המשתמש לא נמצא - מחזיר NotFound

GetGenresBatch (POST):
- מקבל רשימת אימיילים
- עובר על כל אימייל ושולף את נתוני ההאזנות
- מחזיר מערך של אובייקטים עם Email ו-Genres
- משמש לאלגוריתם KNN שצריך להשוות בין משתמשים רבים


שאילתת SQL לשליפת נתוני ז'אנרים:

public async Task<List<ReplayInfo>> GetUserGenres(string email)
{
    string query = $@"Select
                        genres.name,
                        replayes.amount
                      From
                        replayes 
                      Inner Join genres On replayes.genreid = genres.id 
                      Inner Join users On replayes.userid = users.id
                      Where
                        users.email = '{email}'
                      Order By
                        genres.name";
    
    List<ReplayInfo> ri = ((List<ReplayInfo>)await SelectAllAsync(query));
    return ri;
}

הסבר:
- מצטרפת לשלוש טבלאות: replayes, genres, users
- מחזירה את שם הז'אנר וכמות ההאזנות
- ממיינת לפי שם הז'אנר (חשוב ל-KNN)
- מחזירה רשימה של ReplayInfo


שימוש ב-API במערכת הראשית:

המערכת הראשית (Tune.site) קוראת ל-API הזה כדי:
1. לקבל את וקטור הז'אנרים של המשתמש המחובר
2. לקבל וקטורי ז'אנרים של כל המשתמשים האחרים
3. להשוות ביניהם באמצעות אלגוריתם KNN
4. למצוא את המשתמשים עם טעם מוזיקלי הכי דומה


סיכום זרימת הנתונים

1. לקוח (דפדפן) → SignalR Hub → שרת
   - משתמש שולח הודעה בצ'אט
   - SignalR מעביר את ההודעה לשרת
   - השרת שומר במסד הנתונים ומפיץ לכל המשתמשים בצ'אט

2. לקוח → Blazor Component → DAL → מסד נתונים
   - משתמש מבצע פעולה (הרשמה, עדכון פרופיל, וכו')
   - הקומפוננט קורא לשכבת DAL
   - DAL מבצעת פעולת CRUD במסד הנתונים
   - התוצאה חוזרת ללקוח

3. לקוח → Custom API → מסד נתונים API
   - המערכת צריכה נתוני מוזיקה
   - קוראת ל-API עם אימייל
   - API שולף נתונים ממסד נתונים נפרד
   - מחזיר וקטור ז'אנרים

4. שרת → Resend API → אימייל
   - משתמש מבקש איפוס סיסמה או אימות
   - השרת יוצר קוד אקראי
   - קורא ל-Resend API
   - Resend שולח אימייל למשתמש
