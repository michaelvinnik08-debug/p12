namespace Moduls
{
    public class Tuners
    {
        public int Id { get; set; }
        public string username { get; set; }
        public string email { get; set; }
        public string password { get; set; }
        public int role { get; set; }
        public DateTime bantime { get; set; }
        public DateTime created { get; set; }
        public string picture { get; set; }
       
        public Tuners() { }
        public Tuners(int Id, string username, string email, string password, string picture)
        {
            this.Id = Id;
            this.username = username;
            this.password = password;
            this.email = email;
            this.picture = picture;
        }
    }
}

הסבר:
מחלקה זו מייצגת משתמש במערכת. היא מכילה את כל המאפיינים הנדרשים:
- Id: מזהה ייחודי של המשתמש
- username: שם המשתמש המוצג
- email: כתובת אימייל לכניסה למערכת
- password: סיסמה מוצפנת
- role: תפקיד (0=משתמש רגיל, 1=מנהל)
- bantime: תאריך חסימה (אם המשתמש חסום)
- created: תאריך יצירת החשבון
- picture: תמונת פרופיל בפורמט Base64

המחלקה כוללת קונסטרוקטור ריק וקונסטרוקטור עם פרמטרים ליצירת אובייקט משתמש חדש.

הסבר:
פעולה זו מבצעת הרשמה של משתמש חדש למערכת:
1. יוצרת Dictionary עם האימייל כפילטר
2. בודקת אם קיים משתמש עם אותו אימייל במסד הנתונים

   - יוצרת Dictionary עם הערכים החדשים
   - מכניסה את המשתמש למסד הנתונים
   - מחזירה את אובייקט המשתמש שנוצר (כולל ה-ID שנוצר אוטומטית)
4. אם כבר קיים - מחזירה null (הרשמה נכשלה)

public async Task<TimeSpan?> InsertGetObjAsync(Reports R)
{
    if (!IsValidCategory(R.report))
        return null;

    TunersDB s = new TunersDB();
    Tuners t = await s.SelectUserID(R.reported);

    if (t == null || t.bantime > DateTime.UtcNow)
        return null;

    Reports existing = await GetRecentReport(R.reporter, R.reported);
    if (existing != null)
    {
        TimeSpan elapsed = DateTime.UtcNow - existing.created_at;
        TimeSpan cooldown = TimeSpan.FromHours(24);
        if (elapsed < cooldown)
            return cooldown - elapsed;
    }

    Dictionary<string, object> fillValues = new Dictionary<string, object>()
    {
        { "reporter", R.reporter },
        { "reported", R.reported },
        { "report", R.report },
        { "created_at", DateTime.UtcNow }
    };

    await base.InsertGetObjAsync(fillValues);
    return null;
}

הרחבות צד שרת


1. הליך א-סינכרוני (Async/Await)

המערכת משתמשת בהרחבה ב-Async/Await לביצוע פעולות א-סינכרוניות. זה מאפשר לשרת לטפל במספר בקשות במקביל מבלי לחסום את ה-Thread הראשי.

דוגמה מ-BaseDB:

protected async Task<List<T>> SelectAllAsync(Dictionary<string, object> parameters)
{
    return await SelectAllAsync("", parameters);
}

protected async Task<int> InsertAsync(Dictionary<string, object> keyAndValue)
{
    string sqlCommand = PrepareInsertQueryWithParameters(keyAndValue);
    return await ExecNonQueryAsync(sqlCommand);
}

private async Task<int> ExecNonQueryAsync(string query)
{
    if (String.IsNullOrEmpty(query))
        return 0;
        
    await PreQueryAsync(query);
    int rowsEffected = 0;
    
    try
    {
        rowsEffected = await cmd.ExecuteNonQueryAsync();
    }
    catch (Exception e)
    {
        System.Diagnostics.Debug.WriteLine(e.Message + "\nsql:" + cmd.CommandText);
    }
    finally
    {
        await PostQueryAsync();
    }
    
    return rowsEffected;
}

הסבר:
1. כל פעולה שמבצעת גישה למסד הנתונים מוגדרת כ-async
2. השימוש ב-await מאפשר לשרת לשחרר את ה-Thread בזמן שהפעולה מתבצעת
3. זה מאפשר לשרת לטפל בבקשות נוספות במקביל
4. כאשר הפעולה מסתיימת, הקוד ממשיך מהנקודה שבה הוא עצר

יתרונות:
- ביצועים טובים יותר - השרת יכול לטפל ביותר בקשות במקביל
- חווית משתמש טובה יותר - הממשק לא נתקע בזמן פעולות ארוכות
- שימוש יעיל יותר במשאבי השרת


2. הגנה מפני SQL Injection

המערכת משתמשת בפרמטרים SQL כדי להגן מפני התקפות SQL Injection. במקום לבנות שאילתות עם ערכים ישירים, המערכת משתמשת ב-Parameters.

דוגמה מ-BaseDB:

private void AddParameterToCommand(string name, object value)
{
    DbParameter p = cmd.CreateParameter();
    p.ParameterName = name;
    p.Value = value;
    cmd.Parameters.Add(p);
}

private string PrepareWhereQueryWithParameters(Dictionary<string, object> parameters)
{
    string where = "WHERE ";
    string AND = "AND";
    
    if (parameters != null && parameters.Count > 0)
    {
        foreach (KeyValuePair<string, object> param in parameters)
        {
            string prm = $"@W{param.Key}";
            where += $"{param.Key}={prm} {AND} ";
            AddParameterToCommand(prm, param.Value);
        }
        where = where.Remove(where.Length - AND.Length - 2);
    }
    else
        where = "";
        
    return where;
}

private string PrepareInsertQueryWithParameters(Dictionary<string, object> fields)
{
    if (fields == null || fields.Count == 0)
        return "";

    string InKey = "(" + string.Join(",", fields.Keys) + ")";
    string InValue = "VALUES(";
    
    for (int i = 0; i < fields.Values.Count; i++)
    {
        string pn = "@" + i;
        InValue += pn + ',';
        AddParameterToCommand(pn, fields.Values.ElementAt(i));
    }
    
    InValue = InValue.Remove(InValue.Length - 1);
    InValue += ")";

    string sqlCommand = $"INSERT INTO {GetTableName()} {InKey} {InValue};";
    return sqlCommand;
}

הסבר:
1. במקום לבנות שאילתה כך:
   SELECT * FROM users WHERE email='user@example.com'
   
2. המערכת בונה שאילתה עם פרמטרים:
   SELECT * FROM users WHERE email=@Wemail
   
3. הערך של האימייל מועבר בנפרד כפרמטר:
   @Wemail = 'user@example.com'

4. MySQL מטפל בפרמטר בצורה בטוחה ומונע הזרקת קוד זדוני

דוגמה לשימוש:

Dictionary<string, object> filter = new Dictionary<string, object>();
filter.Add("email", userInput);
filter.Add("password", passwordInput);
List<Tuners> t = await SelectAllAsync(filter);

גם אם userInput מכיל תווים מסוכנים כמו ' OR '1'='1, הם יטופלו כטקסט רגיל ולא כקוד SQL.

יתרונות:
- הגנה מלאה מפני SQL Injection
- קוד נקי וקריא
- ביצועים טובים (MySQL יכול לשמור תוכניות ביצוע)
