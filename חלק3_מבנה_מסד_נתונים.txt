        longtext picture "NULL"
    }
    
    chats {
        int id PK "AUTO_INCREMENT"
        int user1_id FK "NOT NULL"
        text last_message "NULL"
        datetime last_message_time "DEFAULT CURRENT_TIMESTAMP"
        int user2_id FK "NOT NULL"
    }
    
    messages {
        int id PK "AUTO_INCREMENT"
        text text "NOT NULL"
        datetime Time "DEFAULT CURRENT_TIMESTAMP"
        int chatid FK "NOT NULL"
        int userid FK "NOT NULL"

    reports {
        int id PK "AUTO_INCREMENT"
        text report "NOT NULL"
        int reported FK "NOT NULL"
        int reporter FK "NOT NULL"
        datetime created_at "DEFAULT CURRENT_TIMESTAMP"
    }
    
    requests {
        int id PK "AUTO_INCREMENT"
        int inviter_id FK "NOT NULL"
        int invited_id FK "NOT NULL"
        datetime matched "DEFAULT CURRENT_TIMESTAMP"
    }

  • Spam or flooding (ספאם או הצפה)
  • Harassment or threats (הטרדה או איומים)
  • Other (אחר)

  • אינדקסים: userid_idx, genreid_idx

הסבר על טבלת הקישור:
טבלה זו מאפשרת למשתמש אחד להיות מקושר לז'אנרים רבים, וכל ז'אנר יכול להיות 
מקושר למשתמשים רבים (קשר Many-to-Many). השדה amount מוסיף מידע נוסף על עוצמת 
הקשר - כמה פעמים המשתמש האזין לז'אנר זה. מידע זה משמש את אלגוריתם ה-KNN 
למציאת משתמשים עם טעם מוזיקלי דומה על ידי השוואת וקטורי הז'אנרים והכמויות.

מנוע מסד הנתונים:
  • מנוע: InnoDB

הערות חשובות:
  1. כל התרשימים לעיל נכתבו בפורמט Mermaid
  2. להמרת התרשימים לתמונות, יש להיכנס לאתר: https://mermaid.live/
  3. יש להעתיק את קוד ה-Mermaid של כל תרשים ולהדביק באתר
  4. האתר יצור תרשים גרפי שניתן להוריד כתמונה PNG או SVG
  5. ניתן להוסיף את התמונות ליומן הפרוייקט
